add_namespace = wa_mp_ita_bld.10

country_event = {
  id = wa_mp_ita_bld.1
  is_triggered_only = yes
  hidden = yes

  immediate = {

    # -------------------------------------------------------------
    # MILITARY FACTORIES -> focus A
    # -------------------------------------------------------------
    if = {
      limit = {
        has_completed_focus = <FOCUS_ID_MIL_FACTORIES>
        NOT = { has_country_flag = wa_mp_ita_115_mil_done }
        115 = { is_controlled_by = ITA }
      }

      115 = {
        add_extra_state_shared_building_slots = 20
        add_building_construction = {
          type = arms_factory
          level = 20
          instant_build = yes
        }
        set_state_flag = wa_mp_ita_115_mil_granted
      }

      set_country_flag = wa_mp_ita_115_mil_done
    }

    # -------------------------------------------------------------
    # CIVILIAN FACTORIES -> focus B
    # -------------------------------------------------------------
    if = {
      limit = {
        has_completed_focus = <FOCUS_ID_CIV_FACTORIES>
        NOT = { has_country_flag = wa_mp_ita_115_civ_done }
        115 = { is_controlled_by = ITA }
      }

      115 = {
        add_extra_state_shared_building_slots = 20
        add_building_construction = {
          type = industrial_complex
          level = 20
          instant_build = yes
        }
        set_state_flag = wa_mp_ita_115_civ_granted
      }

      set_country_flag = wa_mp_ita_115_civ_done
    }

    # -------------------------------------------------------------
    # DOCKYARDS -> focus C
    # -------------------------------------------------------------
    if = {
      limit = {
        has_completed_focus = <FOCUS_ID_DOCKYARDS>
        NOT = { has_country_flag = wa_mp_ita_115_dock_done }
        115 = { is_controlled_by = ITA }
      }

      115 = {
        add_extra_state_shared_building_slots = 15
        add_building_construction = {
          type = dockyard
          level = 15
          instant_build = yes
        }
        set_state_flag = wa_mp_ita_115_dock_granted
      }

      set_country_flag = wa_mp_ita_115_dock_done
    }

    # -------------------------------------------------------------
    # Re-schedule weekly until all 3 are applied
    # -------------------------------------------------------------
    if = {
      limit = {
        OR = {
          NOT = { has_country_flag = wa_mp_ita_115_mil_done }
          NOT = { has_country_flag = wa_mp_ita_115_civ_done }
          NOT = { has_country_flag = wa_mp_ita_115_dock_done }
        }
      }
      country_event = { id = wa_mp_ita_bld.1 days = 10 }
    }
  }
}
