add_namespace = wa_mp_fix_ita_bld

country_event = {
  id = wa_mp_fix_ita_bld.1
  is_triggered_only = yes
  hidden = yes

  immediate = {

    # -------------------------------------------------------------
    # MILITARY FACTORIES -> ITA_italian_army
    # -------------------------------------------------------------
    if = {
      limit = {
        has_completed_focus = ITA_italian_army
        NOT = { has_country_flag = wa_mp_fix_ita_115_mil_done }
        115 = { is_controlled_by = ITA }
      }

      115 = {
        add_extra_state_shared_building_slots = 20
        add_building_construction = {
          type = arms_factory
          level = 20
          instant_build = yes
        }
        set_state_flag = wa_mp_fix_ita_115_mil_granted
      }

      set_country_flag = wa_mp_fix_ita_115_mil_done
    }

    # -------------------------------------------------------------
    # CIVILIAN FACTORIES -> ITA_italian_highways
    # -------------------------------------------------------------
    if = {
      limit = {
        has_completed_focus = ITA_italian_highways
        NOT = { has_country_flag = wa_mp_fix_ita_115_civ_done }
        115 = { is_controlled_by = ITA }
      }

      115 = {
        add_extra_state_shared_building_slots = 20
        add_building_construction = {
          type = industrial_complex
          level = 20
          instant_build = yes
        }
        set_state_flag = wa_mp_fix_ita_115_civ_granted
      }

      set_country_flag = wa_mp_fix_ita_115_civ_done
    }

    # -------------------------------------------------------------
    # DOCKYARDS -> ITA_italian_navy
    # -------------------------------------------------------------
    if = {
      limit = {
        has_completed_focus = ITA_italian_navy
        NOT = { has_country_flag = wa_mp_fix_ita_115_dock_done }
        115 = { is_controlled_by = ITA }
      }

      115 = {
        add_extra_state_shared_building_slots = 15
        add_building_construction = {
          type = dockyard
          level = 15
          instant_build = yes
        }
        set_state_flag = wa_mp_fix_ita_115_dock_granted
      }

      set_country_flag = wa_mp_fix_ita_115_dock_done
    }

    # -------------------------------------------------------------
    # Re-schedule weekly until all 3 are applied
    # -------------------------------------------------------------
    if = {
      limit = {
        OR = {
          NOT = { has_country_flag = wa_mp_fix_ita_115_mil_done }
          NOT = { has_country_flag = wa_mp_fix_ita_115_civ_done }
          NOT = { has_country_flag = wa_mp_fix_ita_115_dock_done }
        }
      }
      country_event = { id = wa_mp_fix_ita_bld.1 days = 10 }
    }
  }
}
